# P17 — Configurable member export implementation plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Shared, reusable staff-created member export templates with ordered column pickers, agreement-state + training-group filters, XLSX/CSV per-run choices, and admin-run direct download.

**Architecture:** A `MemberExportTemplate` model stores ordered JSON column keys and optional filters. A server-side column registry in `apps/members/exports.py` maps stable keys to pure readers. `apps/members/export_templates.py` owns the template service (`build_template_member_queryset`, `render_member_export`). `apps/core/export.py` owns cell preparation and response writers. Admin admin-wrapped run page delivers a CSRF-protected direct download.

**Tech Stack:** Python 3.12+, Django 5.x, `openpyxl` (new production dep), pytest/pytest-django, unittest.mock.

```python
def xlsx_response(*, filename: str, columns: Sequence[str], rows: Iterable[Sequence[Any]]) -> HttpResponse:
    """Render an in-memory XLSX response with bounded column widths."""
    from io import BytesIO
    from openpyxl import Workbook
    from openpyxl.styles import Font
    from openpyxl.utils import get_column_letter

    wb = Workbook()
    ws = wb.active
    ws.append(list(columns))
    header_font = Font(bold=True)
    for cell in ws[1]:
        cell.font = header_font

    max_col_width = 40
    for row in rows:
        ws.append([prepare_export_cell(cell) for cell in row])

    for col_idx, col_name in enumerate(columns, start=1):
        ws.column_dimensions[get_column_letter(col_idx)].width = min(
            max(len(col_name) * 1.5, 10), max_col_width
        )

    buf = BytesIO()
    wb.save(buf)
    buf.seek(0)
    resp = HttpResponse(
        buf.read(),
        content_type="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
    )
    resp["Content-Disposition"] = f'attachment; filename="{filename}"'
    return resp

class MemberExportTemplate(TimeStampedModel):
    """Staff-created member export template with ordered column selection and filters."""

    name = models.CharField(max_length=128)
    column_keys = models.JSONField()
    agreement_status_filters = models.JSONField(default=list, blank=True)
    training_groups = models.ManyToManyField("TrainingGroup", blank=True, related_name="export_templates")
    created_by = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        null=True,
        blank=True,
        on_delete=models.SET_NULL,
        related_name="member_export_templates_created",
    )

    def __str__(self):
        return self.name

    def clean(self):
        super().clean()
        # Local import to avoid circular import (models → exports → models)
        from apps.members.exports import COLUMN_REGISTRY, VALID_AGREEMENT_STATES

        # Validate column_keys
        keys = self.column_keys
        if not isinstance(keys, list) or len(keys) == 0:
            raise ValidationError({"column_keys": "Column list cannot be empty."})
        for key in keys:
            if not isinstance(key, str):
                raise ValidationError({"column_keys": "All column keys must be strings."})
        invalid_keys = [k for k in keys if k not in COLUMN_REGISTRY]
        if invalid_keys:
            raise ValidationError({"column_keys": f"Invalid column keys: {', '.join(invalid_keys)}"})
        if len(keys) != len(set(keys)):
            raise ValidationError({"column_keys": "Column keys must be unique."})

        # Validate agreement_status_filters
        states = self.agreement_status_filters
        if not isinstance(states, list):
            raise ValidationError({"agreement_status_filters": "Must be a list."})
        for state in states:
            if not isinstance(state, str):
                raise ValidationError({"agreement_status_filters": "All state codes must be strings."})
        invalid_states = [s for s in states if s not in VALID_AGREEMENT_STATES]
        if invalid_states:
            raise ValidationError({"agreement_status_filters": f"Invalid agreement states: {', '.join(invalid_states)}"})
        if len(states) != len(set(states)):
            raise ValidationError({"agreement_status_filters": "Agreement states must be unique."})
```

from __future__ import annotations

from dataclasses import dataclass
from typing import Literal

from django.db.models import Prefetch, QuerySet
from django.http import HttpResponse
from django.utils import timezone

from apps.agreements.models import Agreement
from apps.core.export import csv_response, xlsx_response
from apps.members.exports import COLUMN_REGISTRY
from apps.members.models import Member, MemberExportTemplate


@dataclass(frozen=True)
class RenderedMemberExport:
    response: HttpResponse
    row_count: int
    sensitive: bool = False


def build_template_member_queryset(template: MemberExportTemplate) -> QuerySet[Member]:
    """Build the filtered Member queryset for a given export template."""
    qs = Member.objects.select_related(
        "guardian", "guardian__parent_account", "training_group"
    ).prefetch_related(
        Prefetch(
            "agreements",
            queryset=Agreement.objects.filter(is_current=True),
            to_attr="_current_export_agreements",
        )
    )

    if template.agreement_status_filters:
        qs = qs.filter(
            agreements__state__in=template.agreement_status_filters,
            agreements__is_current=True,
        ).distinct()

    if template.training_groups.exists():
        qs = qs.filter(
            training_group__in=template.training_groups.all()
        ).distinct()

    return qs


def render_member_export(
    template: MemberExportTemplate,
    *,
    fmt: Literal["csv", "xlsx"],
) -> RenderedMemberExport:
    """Render a member export for the given template in the specified format."""
    qs = build_template_member_queryset(template)
    columns = [COLUMN_REGISTRY[k].label for k in template.column_keys]
    rows = []
    for member in qs:
        row = []
        for key in template.column_keys:
            spec = COLUMN_REGISTRY[key]
            row.append(spec.reader(member))
        rows.append(row)

    if fmt == "csv":
        response = csv_response(
            filename=f"member-export-{template.pk}-{timezone.localtime().strftime('%Y%m%d-%H%M')}.csv",
            columns=columns,
            rows=rows,
        )
    elif fmt == "xlsx":
        response = xlsx_response(
            filename=f"member-export-{template.pk}-{timezone.localtime().strftime('%Y%m%d-%H%M')}.xlsx",
            columns=columns,
            rows=rows,
        )
    else:
        raise ValueError(f"unsupported format: {fmt}")

    sensitive = any(COLUMN_REGISTRY[k].sensitive for k in template.column_keys)

    return RenderedMemberExport(
        response=response,
        row_count=len(rows),
        sensitive=sensitive,
    )

class OrderedColumnKeysWidget(forms.widgets.Widget):
    """Column picker widget: available columns + add/remove + move-up/move-down.

    Renders all needed markup from Python (no separate template file).
    Serializes only selected stable keys to its hidden JSON input.
    """

    def get_context(self, name, value, attrs):
        ctx = super().get_context(name, value, attrs)
        selected = [
            key for key in value
            if isinstance(key, str) and key in COLUMN_REGISTRY
        ] if isinstance(value, list) else []
        available_keys = [k for k in COLUMN_REGISTRY.keys() if k not in selected]
        ctx["widget"]["available_keys"] = available_keys
        ctx["widget"]["selected"] = selected
        return ctx

    def format_value(self, value):
        if isinstance(value, list):
            return value
        return []

    def render(self, name, value, attrs=None, renderer=None):
        selected = [
            key for key in self.format_value(value)
            if isinstance(key, str) and key in COLUMN_REGISTRY
        ]
        available_keys = [k for k in COLUMN_REGISTRY.keys() if k not in selected]
        ctx = self.get_context(name, selected, attrs)

        parts = ['<div class="fk-column-picker" data-export-column-picker>']

        # Selected columns with move controls
        parts.append('<ul class="fk-column-list" data-column-keys>')
        for key in selected:
            label = COLUMN_REGISTRY[key].label
            parts.append(
                f'<li class="fk-column-item" data-key="{key}">'
                f'<span class="fk-column-label">{label}</span>'
                f'<button type="button" class="fk-column-move-up" data-direction="up" title="Augšup">↑</button>'
                f'<button type="button" class="fk-column-move-down" data-direction="down" title="Lejup">↓</button>'
                f'<button type="button" class="fk-column-remove" data-key="{key}" title="Noņemt">✕</button>'
                f'</li>'
            )
        parts.append('</ul>')

        # Available columns with add controls
        parts.append('<ul class="fk-column-available" data-column-available>')
        for key in available_keys:
            label = COLUMN_REGISTRY[key].label
            parts.append(
                f'<li class="fk-column-available-item">'
                f'<span class="fk-column-label">{label}</span>'
                f'<button type="button" class="fk-column-add" data-key="{key}" data-label="{label}" title="Pievienot">+</button>'
                f'</li>'
            )
        parts.append('</ul>')

        # Hidden JSON input
        parts.append(
            f'<input type="hidden" name="{name}" id="id_{name}"'
            f' value=\'{self._json_value(selected)}\'>'
        )
        parts.append('</div>')

        return "\n".join(parts)

    def _json_value(self, value):
        import json
        return json.dumps(value)

class MemberExportTemplateAdmin(admin.ModelAdmin):
    form = MemberExportTemplateAdminForm
    list_display = ("name", "created_at", "updated_at")
    list_filter = ("created_at",)
    search_fields = ("name",)
    readonly_fields = ("created_by", "created_at", "updated_at")

    def has_view_permission(self, request, obj=None):
        return request.user.is_active and request.user.is_staff

    def has_add_permission(self, request):
        return request.user.is_active and request.user.is_staff

    def has_change_permission(self, request, obj=None):
        return request.user.is_active and request.user.is_staff

    def has_delete_permission(self, request, obj=None):
        return request.user.is_active and request.user.is_staff

    def has_module_permission(self, request):
        return request.user.is_active and request.user.is_staff

    def get_urls(self):
        urls = super().get_urls()
        custom_urls = [
            path(
                "<int:object_id>/run/",
                self.admin_site.admin_view(self.run_view),
                name="members_memberexporttemplate_run",
            ),
        ]
        return custom_urls + urls

    def run_view(self, request, object_id):
        template = get_object_or_404(MemberExportTemplate, pk=object_id)
        if not self.has_view_permission(request, template):
            raise PermissionDenied

        if request.method == "POST":
            form = MemberExportRunForm(request.POST)
            if form.is_valid():
                fmt = form.cleaned_data["fmt"]
                result = render_member_export(template, fmt)

                record_audit_event(
                    action=str(AuditEvent.Action.MEMBER_EXPORT_RUN),
                    actor=request.user,
                    request=request,
                    target_type="member_export_template",
                    target_id=str(template.pk),
                    target_repr="Member export template",
                    metadata={
                        "template_id": template.pk,
                        "column_keys": template.column_keys,
                        "agreement_status_filters": template.agreement_status_filters,
                        "training_group_ids": list(template.training_groups.values_list("pk", flat=True)),
                        "row_count": result.row_count,
                        "format": fmt,
                        "sensitive": result.sensitive,
                    },
                )

                return result.response
        else:
            form = MemberExportRunForm(initial={"fmt": "xlsx"})

        from django.shortcuts import render
        return render(
            request,
            "admin/members/memberexporttemplate/run.html",
            {
                "template": template,
                "form": form,
                "title": f"Eksportēt: {template.name}",
                "opts": self.model._meta,
            },
        )

    def save_model(self, request, obj, form, change):
        audited_fields = {
            "name",
            "column_keys",
            "agreement_status_filters",
            "training_groups",
        }
        operation = ""
        if not change:
            obj.created_by = request.user
            operation = "create"
        elif audited_fields.intersection(form.changed_data):
            operation = "edit"

                target_type="member_export_template",
                target_id=str(obj.pk),
                target_repr="Member export template",
                metadata={"template_id": obj.pk, "operation": operation},
            )

            target_repr="Member export template",
            metadata={"template_id": obj.pk, "operation": "delete"},

                target_id=str(obj.pk),
                target_repr="Member export template",
                metadata={"template_id": obj.pk, "operation": "delete"},
            )
        super().delete_queryset(request, queryset)

document.addEventListener("DOMContentLoaded", function () {
    document.querySelectorAll("[data-export-column-picker]").forEach(function (root) {
        var selected = root.querySelector("[data-column-keys]");
        var available = root.querySelector("[data-column-available]");
        var hidden = root.querySelector("#id_column_keys");
        if (!selected || !available || !hidden) return;

        function button(className, text, title) {
            var element = document.createElement("button");
            element.type = "button";
            element.className = className;
            element.textContent = text;
            element.title = title;
            return element;
        }

        function selectedItem(key, label) {
            var item = document.createElement("li");
            item.className = "fk-column-item";
            item.dataset.key = key;
            var text = document.createElement("span");
            text.className = "fk-column-label";
            text.textContent = label;
            item.append(text);
            var up = button("fk-column-move-up", "↑", "Augšup");
            up.dataset.direction = "up";
            var down = button("fk-column-move-down", "↓", "Lejup");
            down.dataset.direction = "down";
            var remove = button("fk-column-remove", "✕", "Noņemt");
            remove.dataset.label = label;
            item.append(up, down, remove);
            return item;
        }

        function availableItem(key, label) {
            var item = document.createElement("li");
            item.className = "fk-column-available-item";
            var text = document.createElement("span");
            text.className = "fk-column-label";
            text.textContent = label;
            var add = button("fk-column-add", "+", "Pievienot");
            add.dataset.key = key;
            add.dataset.label = label;
            item.append(text, add);
            return item;
        }

        function sync() {
            hidden.value = JSON.stringify(Array.from(
                selected.querySelectorAll(".fk-column-item"),
                function (item) { return item.dataset.key; },
            ));
        }

        root.addEventListener("click", function (event) {
            var add = event.target.closest(".fk-column-add");
            if (add) {
                event.preventDefault();
                selected.append(selectedItem(add.dataset.key, add.dataset.label));
                add.closest(".fk-column-available-item").remove();
                sync();
                return;
            }
            var remove = event.target.closest(".fk-column-remove");
            if (remove) {
                event.preventDefault();
                var item = remove.closest(".fk-column-item");
                available.append(availableItem(item.dataset.key, remove.dataset.label));
                item.remove();
                sync();
                return;
            }
            var move = event.target.closest("[data-direction]");
            if (!move) return;
            event.preventDefault();
            var moving = move.closest(".fk-column-item");
            var sibling = move.dataset.direction === "up"
                ? moving.previousElementSibling
                : moving.nextElementSibling;
            if (sibling) {
                move.dataset.direction === "up"
                    ? selected.insertBefore(moving, sibling)
                    : selected.insertBefore(sibling, moving);
                sync();
            }
        });
